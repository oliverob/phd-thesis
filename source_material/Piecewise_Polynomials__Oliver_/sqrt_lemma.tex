\begin{lemma}
\label{lemma:sqrtpoly}
For any $n \in \mathbb{Z}^+$, there exists a $n$-piecewise polynomial $p(x)$ of maximum degree $d$ such that 
$$|p(x) - \sqrt{x}| \leq \frac{2^{-(2d+1)}(2d)!}{\sqrt{2}(d+1)!d!} = O(d^{-\frac{3}{2}}) \>\>\forall\>\> x\in [2^{-n}, 1)$$
The $n$ polynomial segments are defined over ranges $[2^{-i-1}, 2^{-i})$ $\forall$ $i \in \mathbb{Z}_n$
\end{lemma}
\begin{proof}
Consider $f(x) = \sqrt{x}$. The Taylor series expansion around $x=a$ is given by:
\begin{equation}
    f(x) = \sum_{k=0}^d \frac{f^{(k)}(a)}{k!}(x-a)^k + \frac{f^{(d+1)}(x^*)}{(d+1)!}(x-a)^{d+1}
\end{equation}
where $x^* \in (a,x)$.
\begin{align}
    f^{(k)}(x) &= \frac{1}{2}(-\frac{1}{2})\cdots (\frac{3}{2}-k)x^{\frac{1}{2}-k} \\
    &= (-1)^{k-1} 2^{-k} 3 \times 5 \times \cdots \times (2k-3) x^{\frac{1}{2}-k} \\
    &= (-1)^{k-1} 2^{-2k+1} \frac{(2k-2)!}{(k-1)!}x^{\frac{1}{2}-k}
\end{align}
Hence we can evaluate the Lagrange Remainder for a $d$-degree Taylor series expansion:
\begin{equation}
    \frac{f^{(d+1)}(x^*)}{(d+1)!}(x-a)^{d+1} = (-1)^{d} 2^{-(2d+1)} \frac{(2d)!}{d!(d+1)!}{x^*}^{-\frac{1}{2}-d}(x-a)^{d+1}
\end{equation}
In each region $[2^{-i-1}, 2^{-i}]$ $\forall$ $i \in \mathbb{Z}_n$ we take the Taylor series expansion about $a=2^{-i-1}$, and hence the error in the region is given by:
\begin{equation}
    E_i = (-1)^{d} 2^{-(2d+1)} \frac{(2d)!}{d!(d+1)!}{x^*}^{-\frac{1}{2}-d}(x-2^{-i-1})^{d+1}
\end{equation}
where $x \in [2^{-i-1}, 2^{-i}]$. Therefore, the largest absolute value the error can take is when $x = 2^{-i}$ and $x^* = 2^{-i-1}$:
\begin{align}
        |E_i| &\leq  2^{-(2d+1)} \frac{(2d)!}{d!(d+1)!}2^{(i+1)(\frac{1}{2}+d)-(i+1)(d+1)}\\
        &=  2^{-(2d+1)} \frac{(2d)!}{d!(d+1)!}2^{-\frac{1}{2}(i+1)}
\end{align}
Therefore, the absolute error across all the regions is upper bounded by:
\begin{equation}
    E \leq 2^{-(2d+1)} \frac{(2d)!}{\sqrt{2}d!(d+1)!}
\end{equation}
By applying Stirling's approximation we can find that
\begin{equation}
    2^{-(2d+1)} \frac{(2d)!}{\sqrt{2}d!(d+1)!} = O(d^{-\frac{3}{2}})
\end{equation}
\end{proof}
\begin{duplicate}
For any $n \in \mathbb{Z}^+$, there exists a $(n+1)$-piecewise polynomial $p(x)$ of maximum degree $d$ such that $\|p(x_i)- \sqrt{x_i}\| = O(d^{-\frac{3}{2}})$ for all $2^n$ equally spaced evaluation points between $0$ and $1-2^{-n}$ ($x_i = i2^{-n}$ $\forall$ $i \in \mathbb{Z}_{2^n}$).
\end{duplicate}
\begin{proof}
    All the evaluation points ($x_i = i2^{-n}$ $\forall$ $i \in \mathbb{Z}_{2^n}$) lie in the $n$ regions considered by Lemma~\ref{lemma:sqrtpoly} with the exception of $i = 0$ which lies in the region $[0, 2^{-n})$. In this region we take the 0-degree polynomial $p(x) = 0$ giving an absolute error of 0.
\end{proof}
